import Foundation
import UIKit
import ImagePlayground

@_silgen_name("TagtagStickerCreationAIFinished")
private func nativeFinished(_ bytes: UnsafeRawPointer?, _ length: Int)

@_silgen_name("TagtagStickerCreationAICancelled")
private func nativeCancelled()

@available(iOS 18.1, *)
@MainActor
private final class PlaygroundSession: NSObject, ImagePlaygroundViewController.Delegate {
    private var controller: ImagePlaygroundViewController?

    func present(from presenter: UIViewController) {
        let controller = ImagePlaygroundViewController()
        controller.delegate = self
        controller.modalPresentationStyle = .fullScreen
        self.controller = controller
        presenter.present(controller, animated: true)
    }

    func abandon() {
        // The Objective-C editor dismisses its navigation controller, including
        // this presented playground. Avoid starting two UIKit dismissals at once.
        controller?.delegate = nil
        controller = nil
        activeSession = nil
    }

    func imagePlaygroundViewController(_ imagePlaygroundViewController: ImagePlaygroundViewController,
                                       didCreateImageAt imageURL: URL) {
        let imageData: Data?
        do {
            imageData = try Data(contentsOf: imageURL)
        } catch {
            let failure = error as NSError
            NSLog("[TagtagImagePlayground] Generated image read failed: %@ (%ld)", failure.domain, failure.code)
            imageData = nil
        }
        // Image Playground supplies a temporary file. Consume and clean it up
        // before dismissal, when its lifetime is still guaranteed.
        try? FileManager.default.removeItem(at: imageURL)
        imagePlaygroundViewController.dismiss(animated: true) {
            guard activeSession === self else { return }
            if let imageData = imageData {
                imageData.withUnsafeBytes { buffer in
                    nativeFinished(buffer.baseAddress, buffer.count)
                }
            } else {
                nativeFinished(nil, 0)
            }
            activeSession = nil
        }
    }

    func imagePlaygroundViewControllerDidCancel(_ imagePlaygroundViewController: ImagePlaygroundViewController) {
        imagePlaygroundViewController.dismiss(animated: true) {
            guard activeSession === self else { return }
            nativeCancelled()
            activeSession = nil
        }
    }
}

@MainActor private var activeSession: AnyObject?

@_cdecl("TagtagStickerCreationAIAvailable")
public func tagtagStickerCreationAIAvailable() -> Int32 {
    if #available(iOS 18.1, *) {
        return ImagePlaygroundViewController.isAvailable ? 1 : 0
    }
    return 0
}

@_cdecl("TagtagStickerCreationAIOpen")
public func tagtagStickerCreationAIOpen(_ presenter: UnsafeMutableRawPointer?) {
    DispatchQueue.main.async {
        guard let presenter,
              let viewController = Unmanaged<UIViewController>.fromOpaque(presenter).takeUnretainedValue() as UIViewController?,
              activeSession == nil else {
            nativeCancelled()
            return
        }
        if #available(iOS 18.1, *), ImagePlaygroundViewController.isAvailable {
            let session = PlaygroundSession()
            activeSession = session
            session.present(from: viewController)
        } else {
            nativeCancelled()
        }
    }
}

@_cdecl("TagtagStickerCreationAICancel")
public func tagtagStickerCreationAICancel() {
    DispatchQueue.main.async {
        if #available(iOS 18.1, *), let session = activeSession as? PlaygroundSession {
            session.abandon()
        }
    }
}
